\renewcommand{\secao}{EMBEDDINGS}
% Slide 3; 45 segundos
\begin{frame}[t]{Um embedding representa o texto em números}
\normalfont\sffamily
\begin{center}
{\large Descrição do produto}\par\vspace{3mm}
$\downarrow\quad f_{\theta}$\par\vspace{3mm}
{\Large $\mathbf{z}=(z_1,z_2,\ldots,z_{768})$}
\end{center}
\par\vspace{3mm}
\begin{itemize}
\item O \textbf{encoder} $f_{\theta}$ transforma textos em vetores de tamanho fixo.
\item Cada dimensão é aprendida; não corresponde a uma coluna como peso ou marca.
\item Textos com comprimentos diferentes podem ser comparados no mesmo espaço.
\end{itemize}
\par\vspace{2mm}
\alert{A qualidade do vetor determina o que o classificador consegue distinguir.}
\fonte{Anggoro et al., seções 3.2 e 4.3.1; explicação didática da representação.}
\end{frame}
\note{\small Embedding é uma representação numérica aprendida. No artigo, toda descrição termina em 768 números. Não se extrai manualmente uma coluna para marca e outra para capacidade: os atributos permanecem no texto e influenciam a representação de maneira distribuída. O símbolo theta representa os parâmetros do encoder, que serão atualizados no ajuste contrastivo. Ter um vetor não garante boa representação: é necessário orientar seu aprendizado para a tarefa.\par [Sources] Anggoro et al., seções 3.2 e 4.3.1; explicação didática da representação.}

\renewcommand{\secao}{BERT E POOLING}
% Slide 4; 55 segundos
\begin{frame}[t]{Pooling reúne os tokens em um único vetor}
\normalfont\sffamily
\textbf{BERT ou DistilBERT} produz um vetor contextual para cada token.
\par\vspace{3mm}
\begin{center}
$h_1,\ h_2,\ \ldots,\ h_m\quad\xrightarrow{\ \text{pooling}\ }\quad\mathbf{z}\in\mathbb{R}^{768}$
\end{center}
\par\vspace{3mm}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}lp{.68\textwidth}@{}}
\toprule
\textbf{Estratégia} & \textbf{Como obtém a representação do texto}\\
\midrule
Média & Calcula a média dos vetores dos tokens.\\
Máximo & Seleciona o maior valor em cada dimensão.\\
{[CLS]} & Usa o vetor do token especial de classificação.\\
\bottomrule
\end{tabular}
\par\vspace{2.5mm}
O estudo monta o SBERT com \alert{encoders já pré-treinados} e ajusta seus parâmetros.
\fonte{Anggoro et al., seções 3.2 e 4.3.1.}
\end{frame}
\note{\small Token é uma unidade de processamento, que pode ser uma palavra ou parte dela. BERT produz representações contextuais: o restante da descrição influencia o vetor de cada token. Pooling transforma uma sequência de comprimento variável em um vetor fixo. Na média, considera-se a agregação dos tokens válidos; no máximo, cada dimensão pode vir de um token diferente. CLS é outra forma de obter uma representação. O estudo utiliza BERT e DistilBERT base uncased pré-treinados, e não treina um transformer desde pesos aleatórios. DistilBERT é a alternativa menor.\par [Sources] Anggoro et al., seções 3.2 e 4.3.1.}

\renewcommand{\secao}{SENTENCE BERT}
% Slide 5; 65 segundos
\begin{frame}[t]{Os dois ramos do SBERT compartilham pesos}
\normalfont\sffamily
\textbf{Rede siamesa / bi-encoder:} cada texto é codificado separadamente.
\par\vspace{2.5mm}
\begin{center}
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{@{}rcl@{}}
Descrição âncora $a$ & $\xrightarrow{\quad f_{\theta}\quad}$ & Vetor $\mathbf{z}_a$\\
Descrição positiva $b$ & $\xrightarrow{\quad f_{\theta}\quad}$ & Vetor $\mathbf{z}_b$\\
\end{tabular}
\end{center}
\par\vspace{2.5mm}
\begin{itemize}
\item Os dois ramos usam \textbf{o mesmo encoder e o mesmo pooling}.
\item A perda compara os vetores e atualiza os parâmetros compartilhados.
\item Depois do ajuste, cada descrição pode gerar seu embedding individualmente.
\end{itemize}
\fonte{Anggoro et al., seção 3.2 e Figura 1.}
\end{frame}
\note{\small Os ramos desenhados representam duas aplicações da mesma função, com os mesmos parâmetros theta. Eles não aprendem independentemente nem recebem um único texto concatenado. A comparação acontece depois que cada descrição produz um vetor. Os gradientes dos pares atualizam os pesos compartilhados. Na inferência, não é preciso fornecer um par positivo: uma descrição passa pelo encoder ajustado e produz o vetor utilizado pelo classificador.\par [Sources] Anggoro et al., seção 3.2 e Figura 1.}

\renewcommand{\secao}{APRENDIZAGEM CONTRASTIVA}
% Slide 6; 65 segundos
\begin{frame}[t]{O código conhecido orienta a aproximação}
\normalfont\sffamily
\textbf{Âncora + outra transação do mesmo código = par positivo.}
\par\vspace{3mm}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}lp{.59\textwidth}c@{}}
\toprule
\textbf{Papel} & \textbf{Descrição ilustrativa} & \textbf{Classe}\\
\midrule
Âncora & Notebook Lenovo ThinkPad, SSD 512 GB & A\\
Positivo & Laptop ThinkPad Lenovo, 512GB & A\\
Outro produto & Monitor LCD para computador & B\\
\bottomrule
\end{tabular}
\par\vspace{3mm}
\begin{itemize}
\item \textbf{Supervisionada:} os rótulos definem quais pares são positivos.
\item \textbf{Contrastiva:} o par positivo é comparado com alternativas.
\item Os pares usam transações reais, preservando o texto original.
\end{itemize}
\fonte{Anggoro et al., seção 3.2 e Figura 2. Exemplo didático; classes A/B hipotéticas.}
\end{frame}
\note{\small O rótulo entra antes da função de perda, ao construir os pares. Positivo não significa texto idêntico: outra transação da mesma classe pode descrever a mercadoria de modo diferente. Os autores sorteiam positivos sem reposição e comparam um e três por âncora. Usar transações existentes evita criar artificialmente uma nova descrição que possa perder especificações técnicas. A classe B é uma alternativa negativa neste exemplo; no treinamento, a MNR obtém as alternativas a partir dos outros pares do lote.\par [Sources] Anggoro et al., seção 3.2 e Figura 2. Exemplo didático; classes A/B hipotéticas.}

\renewcommand{\secao}{SIMILARIDADE}
% Slide 7; 55 segundos
\begin{frame}[t]{O cosseno compara a direção dos vetores}
\normalfont\sffamily
\begin{center}
{\Large $\operatorname{sim}(\mathbf{u},\mathbf{v})=
\dfrac{\mathbf{u}^{\mathsf T}\mathbf{v}}{\|\mathbf{u}\|\,\|\mathbf{v}\|}$}
\end{center}
\par\vspace{3mm}
\begin{columns}[T,onlytextwidth]
\begin{column}{.47\textwidth}
\textbf{Similaridade de cosseno}\par\vspace{2mm}
Compara a direção, normalizando o comprimento dos vetores.\par\vspace{2mm}
Direções próximas: cosseno próximo de 1.
\end{column}
\begin{column}{.47\textwidth}
\textbf{Produto escalar}\par\vspace{2mm}
Usa $\mathbf{u}^{\mathsf T}\mathbf{v}$ sem normalização.\par\vspace{2mm}
Depende da direção e também da magnitude.
\end{column}
\end{columns}
\par\vspace{3mm}
\alert{Similaridade é um escore entre textos; não é a probabilidade de um código.}
\fonte{Anggoro et al., seção 3.3; formulação didática da similaridade de cosseno.}
\end{frame}
\note{\small O cosseno normaliza o produto escalar pelas normas dos vetores. Para vetores não nulos, varia de menos um a um; direções próximas tendem a produzir escores maiores. O produto escalar também depende do comprimento dos vetores. O artigo testa ambas as opções. A similaridade sozinha não atribui um código nem fornece uma probabilidade calibrada. Na próxima etapa, a softmax converte os escores em uma distribuição sobre candidatos do lote, não sobre toda a nomenclatura fiscal.\par [Sources] Anggoro et al., seção 3.3; formulação didática da similaridade de cosseno.}

\renewcommand{\secao}{MULTIPLE NEGATIVE RANKING LOSS}
% Slide 8; 75 segundos
\begin{frame}[t]{Cada positivo também é candidato no lote}
\normalfont\sffamily
\textbf{Lote com três pares:} $(a_1,b_1)$, $(a_2,b_2)$ e $(a_3,b_3)$.
\par\vspace{2.5mm}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}lccc@{}}
\toprule
\textbf{Similaridade} & $b_1$ & $b_2$ & $b_3$\\
\midrule
$a_1$ & \textcolor{Verde}{\textbf{0,80}} & 0,30 & 0,10\\
$a_2$ & 0,20 & \textcolor{Verde}{\textbf{0,75}} & 0,15\\
$a_3$ & 0,10 & 0,25 & \textcolor{Verde}{\textbf{0,85}}\\
\bottomrule
\end{tabular}
\end{center}
\par\vspace{3mm}
\begin{itemize}
\item \textcolor{Verde}{\textbf{Diagonal:}} positivo correspondente a cada âncora.
\item \textbf{Fora da diagonal:} negativos obtidos dos outros pares.
\item A MNR favorece a diagonal diante das alternativas.
\end{itemize}
\par\vspace{2mm}
{\small Exemplo ilustrativo: três pares de classes distintas.}
\fonte{Anggoro et al., seção 3.3 e Apêndice II. Matriz numérica didática.}
\end{frame}
\note{\small A MNR não recebe apenas uma distância entre dois textos. Para n pares, constrói n por n escores. Na primeira linha, b1 é o positivo de a1; b2 e b3 funcionam como negativos para a1, embora sejam positivos de suas próprias âncoras. As posições diagonais são os alvos. Os valores são didáticos e não resultados do artigo. O método aproveita os negativos do próprio lote sem exigir sua seleção manual. Se dois pares tiverem o mesmo código, pode ocorrer um falso negativo; o manuscrito não detalha um mecanismo para evitar isso. Essa ressalva será retomada nos limites.\par [Sources] Anggoro et al., seção 3.3 e Apêndice II. Matriz numérica didática.}

\renewcommand{\secao}{DO ESCORE À PERDA}
% Slide 9; 85 segundos
\begin{frame}[t]{A MNR penaliza a baixa preferência pelo positivo}
\normalfont\sffamily
\textbf{Primeira linha:} positivo $b_1$ com escores $(0{,}80;\ 0{,}30;\ 0{,}10)$.
\par\vspace{2.5mm}
\begin{enumerate}
\item \textbf{Escala:} com $s=2$, os escores viram $(1{,}60;\ 0{,}60;\ 0{,}20)$.
\item \textbf{Softmax:} transforma esses escores em preferências relativas.
\end{enumerate}
\par\vspace{1mm}
\[
p(b_1\mid a_1)=\frac{e^{1{,}60}}{e^{1{,}60}+e^{0{,}60}+e^{0{,}20}}
\approx \textcolor{Verde}{0{,}619}
\]
\begin{enumerate}
\setcounter{enumi}{2}
\item \textbf{Perda:} $-\log(0{,}619)\approx 0{,}479$; o ajuste busca reduzi-la.
\end{enumerate}
\par\vspace{3mm}
\alert{Maior preferência pelo positivo $\Rightarrow$ menor perda.}
\par\vspace{2mm}
{\small Escala 2 apenas para ilustrar o cálculo; o artigo avalia 20 e 50.}
\fonte{Anggoro et al., seção 3.3 e Apêndice II; cálculo didático da MNR padrão.}
\end{frame}
\note{\small O exemplo continua a primeira linha do slide anterior. Multiplicamos as similaridades por uma escala e aplicamos softmax. O positivo recebe aproximadamente 61,9 por cento da massa entre os três candidatos. A entropia cruzada para essa linha é o negativo do logaritmo natural dessa probabilidade, cerca de 0,479. Calcula-se a perda de cada linha e sua média; a retropropagação atualiza o encoder para reduzi-la. As probabilidades são relativas ao lote e não expressam a chance de correção fiscal. A escala 2 simplifica a ilustração; 20 e 50 são os valores testados no artigo. A fórmula completa e a versão simétrica estão no complemento.\par [Sources] Anggoro et al., seção 3.3 e Apêndice II; cálculo didático da MNR padrão.}

\renewcommand{\secao}{TREINAMENTO E INFERÊNCIA}
% Slide 10; 60 segundos
\begin{frame}[t]{Aprender o vetor e prever o código são duas etapas}
\normalfont\sffamily
\begin{columns}[T,onlytextwidth]
\begin{column}{.47\textwidth}
\textcolor{Verde}{\textbf{1. Ajustar o SBERT}}\par\vspace{2mm}
\textbf{70\% das transações}\par\vspace{3mm}
Descrições + códigos\par\vspace{1mm}
$\downarrow$\par\vspace{1mm}
Pares positivos + MNR\par\vspace{1mm}
$\downarrow$\par\vspace{1mm}
Encoder ajustado ao domínio
\end{column}
\begin{column}{.47\textwidth}
\textcolor{Verde}{\textbf{2. Treinar o classificador}}\par\vspace{2mm}
\textbf{30\% das transações}\par\vspace{3mm}
Embeddings do SBERT ajustado\par\vspace{1mm}
$\downarrow$\par\vspace{1mm}
SVM ou Random Forest\par\vspace{1mm}
$\downarrow$\par\vspace{1mm}
Validação estratificada: 5 folds
\end{column}
\end{columns}
\par\vspace{3mm}
\textbf{Nova descrição:}\par\vspace{2mm}
Texto $\rightarrow$ SBERT ajustado $\rightarrow$ vetor $\rightarrow$ classificador $\rightarrow$ código
\fonte{Anggoro et al., seção 3.1, seção sobre classificadores e seção 5.1.}
\end{frame}
\note{\small A MNR treina o gerador de embeddings, não o classificador fiscal final. O artigo reserva 70 por cento das transações para essa etapa. Os outros 30 por cento são usados para treinar e validar SVM ou Random Forest com cinco partições estratificadas. Não se trata de reservar 30 por cento exclusivamente para teste final. Na segunda etapa, o SBERT ajustado gera as features: SVM aprende fronteiras de decisão e Random Forest combina árvores. Na inferência não se conhece o código nem se precisa de um positivo: a nova descrição percorre a cadeia mostrada no rodapé.\par [Sources] Anggoro et al., seção 3.1, seção sobre classificadores e seção 5.1.}

Precisão mede quantas previsões de uma classe estão corretas; revocação mede quantos exemplos da classe foram recuperados. F1 combina as duas. A média ponderada dá mais peso às classes frequentes. A estratificação preserva proporções, sem corrigir o desbalanceamento. O artigo também relata kappa e médias micro.\par [Sources] Anggoro et al., seções 3.1 e 5.1; Tabelas I e III.}

\begin{itemize}
\item \textbf{LIME:} uma explicação local indica influência de atributos técnicos.
\item A evidência vale para a limpeza testada; não condena toda normalização.

\end{frame}
\note{\small A limpeza testada inclui minúsculas, remoção de duplicatas, tratamento de caracteres alfanuméricos e tokens de ao menos dois caracteres. Ela reduz o total de tokens, mas pode eliminar capacidade, modelo e dimensões. A Tabela IV mostra quedas com SVM e RF. LIME explica localmente uma predição por perturbações da entrada; a Figura 3 ilustra o papel de 512 GB, com redução da probabilidade da classe principal de cerca de 0,82 para 0,71 após a limpeza. Isso não demonstra causalidade geral nem separa o efeito de cada operação. A ligação conceitual é que preservar o sinal relevante importa tanto quanto escolher o encoder.\par [Sources] Anggoro et al., seção 3.1, Tabela IV e Figura 3.}

\renewcommand{\secao}{LIMITES}
% Slide 15; 30 segundos
\begin{frame}[t]{A qualidade dos pares limita o aprendizado}
\normalfont\sffamily
\begin{itemize}
\item \textbf{Rótulos declarados:} um código incorreto pode formar um par inadequado.
\item \textbf{Negativos do lote:} outro par do mesmo código pode gerar um falso negativo.
\item \textbf{Generalização:} dois países e dois capítulos não cobrem toda a nomenclatura.
\item \textbf{Custo:} ajustar o encoder e armazenar o classificador exige recursos.
\end{itemize}
\par\vspace{3mm}
\alert{Proximidade no espaço vetorial não assegura conformidade com regras fiscais.}
\par\vspace{3mm}
{\small Rótulos e falsos negativos: leitura crítica do método; efeitos não medidos no artigo.}
\fonte{Anggoro et al., seções 3 e 5, Tabela III e conclusão; leitura crítica do protocolo.}
\end{frame}
\note{\small O primeiro limite é a supervisão: os códigos declarados são tratados como referência, sem auditoria independente de todos os registros descrita no artigo. Outro limite é tratar pares diferentes como negativos; a mesma classe pode reaparecer no lote. O manuscrito não documenta como impede isso. A avaliação não demonstra transferência entre países, robustez temporal ou desempenho em português. O custo de embeddings é reconhecido pelos autores, e a Tabela III mostra que a cadeia pode ficar maior. A ausência de regras legais explícitas impede interpretar o resultado como garantia de conformidade fiscal. Estas observações delimitam a evidência, sem afirmar que foram medidas experimentalmente.\par [Sources] Anggoro et al., seções 3 e 5, Tabela III e conclusão; leitura crítica do protocolo.}

\renewcommand{\secao}{IDEIA CENTRAL}
% Slide 16; 20 segundos
\begin{frame}[t]{O rótulo ensina o modelo a comparar descrições}
\normalfont\sffamily
\par\vspace{3mm}
\begin{enumerate}
\item \textbf{SBERT} transforma cada descrição em um embedding.
\item \textbf{Os códigos conhecidos} definem os pares positivos.
\item \textbf{A MNR} favorece esses pares diante dos candidatos do lote.
\item \textbf{SVM ou Random Forest} convertem os embeddings em previsões.
\end{enumerate}
\par\vspace{3mm}
{\large\color{Verde}\bfseries Aprender uma representação adequada ao domínio\\melhora a classificação nas bases estudadas.\par}
\fonte{Anggoro et al., Figura 1, Tabela III e conclusão.}
\end{frame}
\note{\small Retomar a pergunta inicial. O conhecimento de classe entra na construção dos pares, orienta o encoder pela MNR e depois volta a ser usado no treinamento do classificador. O principal aprendizado é que a representação merece ser treinada para o domínio, preservando a informação técnica do texto. Essa é a contribuição ilustrada pelos resultados de Anggoro. A hipótese pode motivar estudos futuros, mas sua eficácia precisa ser validada em cada contexto.\par [Sources] Anggoro et al., Figura 1, Tabela III e conclusão.}

\renewcommand{\secao}{COMPLEMENTO}
% Slide 17; complemento
\begin{frame}[t]{A MNR é uma entropia cruzada sobre o lote}
\normalfont\sffamily
Para um lote de $n$ pares positivos $(a_i,b_i)$:
\[
\mathcal{L}_{\rightarrow}
=-\frac{1}{n}\sum_{i=1}^{n}
\log\frac{\exp\!\left(s\,\operatorname{sim}(f_{\theta}(a_i),f_{\theta}(b_i))\right)}
{\sum_{j=1}^{n}\exp\!\left(s\,\operatorname{sim}(f_{\theta}(a_i),f_{\theta}(b_j))\right)}.
\]
\par\vspace{2mm}
\begin{itemize}
\item $s$: escala das similaridades; o alvo de cada linha é o índice $i$.
\item O denominador inclui o positivo e os demais candidatos.
\item A escala altera a concentração da softmax; não muda a ordem dos escores.
\end{itemize}
\par\vspace{3mm}
\textbf{MNR simétrica:}\quad
$\mathcal{L}_{\mathrm{sim}}=\frac{1}{2}\left(\mathcal{L}_{\rightarrow}+\mathcal{L}_{\leftarrow}\right)$.
\par\vspace{2mm}
{\small No sentido inverso, cada positivo busca sua âncora correspondente.}
\fonte{Anggoro et al., seção 3.3, seção 4.3.2 e Apêndice II; formulação didática da MNR padrão.}
\end{frame}
\note{\small Esta é a expressão didática da MNR padrão, consistente com a descrição de comparações do manuscrito. Ela não é uma transcrição literal da linha de pseudocódigo do Apêndice II, cuja notação é simplificada. Os alvos são índices da diagonal e a entropia cruzada penaliza a baixa preferência pelo positivo correspondente. A escala corresponde ao inverso da temperatura na parametrização usual. A versão simétrica calcula a média das perdas nas duas direções. A matriz de similaridade é transposta para o sentido inverso, respeitando o alvo diagonal.\par [Sources] Anggoro et al., seção 3.3, seção 4.3.2 e Apêndice II; formulação didática da MNR padrão.}
